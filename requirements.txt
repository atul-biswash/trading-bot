# Exchange connectivity. python-binance is chosen over the lighter official
# connector for its built-in Testnet support, threaded/async websocket
# managers, and mature order helpers.
python-binance==1.0.37

# Data / numerical stack.
# NOTE — `pandas-ta` was intentionally REMOVED. The 0.3.x line this project
# originally targeted has been pulled from PyPI, and the only remaining release
# (0.4.x) forces a heavyweight `numba`/LLVM dependency plus pinned NumPy/pandas
# versions — fragile for software that manages money. Indicators (SMA, EMA, RSI,
# MACD, Bollinger, ATR, ...) are implemented as small, unit-tested functions in
# `indicators/` instead. To re-add the library: `pandas-ta==0.4.71b0` together
# with `numpy>=2.2.6,<2.3` and `pandas>=2.3.2` (this also pulls in numba).
numpy==2.5.1   # verified behaviour: numpy.float64 is a float subclass, which is
               # what the Money guard's _reject_float catches on the realistic
               # leak path (DataFrame.iloc). Not a routine bump.
pandas==2.3.3  # verified behaviour: float64 OHLCV, NaN propagation through
               # ewm/rolling (the indicator warmup contract), and the iloc return
               # type the one deliberate Decimal->float boundary rests on.
               # pandas itself constrains numpy only as `>=1.26.0` — the upper
               # bound above is ours, not inherited. Not a routine bump.

# Configuration & validation.
pydantic==2.13.4  # the most semantically load-bearing pin here: the Money guard
                  # IS pydantic validation (_reject_float runs as a
                  # BeforeValidator), and Decimal coercion behaviour —
                  # shortest-repr, verified for this version — is not
                  # contractual across releases. Transitively pins
                  # pydantic-core==2.46.4, the engine doing the coercion.
                  # Not a routine bump.
pydantic-settings==2.14.2
PyYAML==6.0.3

# Resilience: retry/backoff for the exchange adapter.
tenacity==9.1.4

# Nicer console logging when available (logger degrades gracefully without it).
rich==15.0.0

# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------
# Unimported and still live. pydantic-settings pulls it in unconditionally and
# executes it whenever `Secrets()` is constructed -- which is the path the three
# opt-in Testnet integration tests gate through, and how a developer's `.env` is
# read at all. Deleting it because grep finds no `import dotenv` would remove a
# dependency the credential path actually runs.
python-dotenv==1.2.2
